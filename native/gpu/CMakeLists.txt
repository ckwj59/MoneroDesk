cmake_minimum_required(VERSION 3.24)
project(XMRIndependentCuda LANGUAGES CXX CUDA)
add_library(xmr-randomx-cuda SHARED xmr_cuda.cu)
set(XMR_VM_LANES 32 CACHE STRING "Cooperating lanes per RandomX VM: 8, 16 or 32")
if(NOT XMR_VM_LANES MATCHES "^(8|16|32)$")
  message(FATAL_ERROR "XMR_VM_LANES must be 8, 16 or 32")
endif()
target_compile_definitions(xmr-randomx-cuda PRIVATE XMR_VM_LANES=${XMR_VM_LANES})
set(XMR_VM_THREADS 128 CACHE STRING "CUDA threads per interpreter block: 32, 64 or 128")
if(NOT XMR_VM_THREADS MATCHES "^(32|64|128)$")
  message(FATAL_ERROR "XMR_VM_THREADS must be 32, 64 or 128")
endif()
target_compile_definitions(xmr-randomx-cuda PRIVATE XMR_VM_THREADS=${XMR_VM_THREADS})
option(XMR_VM_CODE_GLOBAL "Keep decoded per-hash programs in device memory instead of shared memory" ON)
if(XMR_VM_CODE_GLOBAL)
  target_compile_definitions(xmr-randomx-cuda PRIVATE XMR_VM_CODE_GLOBAL)
endif()
set_target_properties(xmr-randomx-cuda PROPERTIES CUDA_ARCHITECTURES "75;86;89;120" CUDA_RUNTIME_LIBRARY Static)
target_compile_options(xmr-randomx-cuda PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:--allow-unsupported-compiler;--prec-div=true;--prec-sqrt=true;--fmad=false>)
target_compile_features(xmr-randomx-cuda PRIVATE cxx_std_17)
option(XMR_CUDA_DIAGNOSTICS "Include local CPU/GPU stage diagnostics; never ship" OFF)
if(XMR_CUDA_DIAGNOSTICS)
  target_compile_definitions(xmr-randomx-cuda PRIVATE XMR_CUDA_DIAGNOSTICS)
  target_link_libraries(xmr-randomx-cuda PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../build/Release/randomx.lib")
endif()
